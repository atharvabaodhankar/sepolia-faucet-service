02 — Capabilities

Everything you need,
nothing you don't.

🌐

Multi-Network

Sepolia ETH and Polygon Amoy MATIC from one unified endpoint. More networks planned on the roadmap.

⚡

Zero-Auth API

No API keys, no sign-up, no dashboard. POST your address and receive tokens. Designed for developer speed.

🛡️

Smart Rate Limiting

Per-address, per-network 24-hour cooldown. Automatic balance checks prevent the faucet from being drained.

🔐

Admin Bypass

Pass a master password to skip all checks instantly — ideal for CI/CD pipelines and automated test suites.

📊

Health Status API

GET /api/status returns live faucet balances, requests remaining, and per-network health in real time.

🔗

Explorer Deep Links

Every success response includes a direct Etherscan or PolygonScan link so you can verify the TX instantly.

useMultiNetworkFaucet.js
import { useState, useCallback } from 'react';

const BASE = 'https://sepolia-faucet-service.vercel.app/api/faucet';

export function useMultiNetworkFaucet() {
  const [loading, setLoading] = useState(false);
  const [result,  setResult]  = useState(null);
  const [error,   setError]   = useState(null);

  const requestTokens = useCallback(async (address, network = 'sepolia') => {
    setLoading(true); setError(null); setResult(null);
    try {
      const res  = await fetch(BASE, {
        method:  'POST',
        headers: { 'Content-Type': 'application/json' },
        body:    JSON.stringify({ address, network })
      });
      const data = await res.json();
      if (!data.success) throw new Error(data.message);
      setResult(data);
      return data;
    } catch (err) {
      setError(err.message); throw err;
    } finally {
      setLoading(false);
    }
  }, []);

  return { requestTokens, loading, result, error };
}

// ─── Usage ────────────────────────────────────────────────
function FundButton({ address }) {
  const { requestTokens, loading, result, error } = useMultiNetworkFaucet();

  return (
    <div>
      <button onClick={() => requestTokens(address, 'polygon')} disabled={loading}>
        {loading ? 'Sending…' : 'Get Polygon MATIC'}
      </button>

      {result && (
        <p>
          ✅ <a href={result.explorerUrl} target="_blank">
            {result.transactionHash.slice(0, 20)}…
          </a>
        </p>
      )}
      {error && <p style={{ color: 'red' }}>❌ {error}</p>}
    </div>
  );
}
faucet_client.py
import requests
from dataclasses import dataclass
from typing import Optional

BASE_URL = "https://sepolia-faucet-service.vercel.app"

@dataclass
class FaucetResult:
    success:          bool
    transaction_hash: str
    amount:           str
    currency:         str
    network:          str
    explorer_url:     str

class MultiNetworkFaucet:
    def __init__(self, master_password: Optional[str] = None):
        self.base_url        = BASE_URL
        self.master_password = master_password

    def request_tokens(self, address: str, network: str = "sepolia") -> FaucetResult:
        payload = {"address": address, "network": network}
        if self.master_password:
            payload["masterPassword"] = self.master_password

        r = requests.post(f"{self.base_url}/api/faucet", json=payload, timeout=30)
        r.raise_for_status()
        d = r.json()
        if not d.get("success"):
            raise Exception(d.get("message", "Unknown faucet error"))

        return FaucetResult(
            success          = d["success"],
            transaction_hash = d["transactionHash"],
            amount           = d["amount"],
            currency         = d["currency"],
            network          = d["network"],
            explorer_url     = d["explorerUrl"],
        )

    def get_status(self) -> dict:
        return requests.get(f"{self.base_url}/api/status", timeout=10).json()

# ─── Usage ────────────────────────────────────────────────────────────
if __name__ == "__main__":
    faucet = MultiNetworkFaucet()

    result = faucet.request_tokens("0xYourAddress", "sepolia")
    print(f"[Sepolia] {result.amount} {result.currency} → {result.explorer_url}")

    result = faucet.request_tokens("0xYourAddress", "polygon")
    print(f"[Polygon] {result.amount} {result.currency} → {result.explorer_url}")
faucet.mjs · Node 18+
const FAUCET = 'https://sepolia-faucet-service.vercel.app/api/faucet';

async function requestTokens(address, network = 'sepolia', masterPassword = null) {
  const body = { address, network };
  if (masterPassword) body.masterPassword = masterPassword;

  const res  = await fetch(FAUCET, {
    method:  'POST',
    headers: { 'Content-Type': 'application/json' },
    body:    JSON.stringify(body)
  });
  const data = await res.json();
  if (!data.success) throw new Error(`Faucet: ${data.message}`);
  return data;
}

// Batch-fund multiple test wallets on Polygon Amoy
async function fundTestAccounts(addresses, network = 'polygon') {
  const results = await Promise.allSettled(
    addresses.map(addr => requestTokens(addr, network))
  );
  results.forEach((r, i) => {
    if (r.status === 'fulfilled')
      console.log(`✅ ${addresses[i]} → ${r.value.transactionHash}`);
    else
      console.error(`❌ ${addresses[i]} → ${r.reason.message}`);
  });
}

await fundTestAccounts([
  '0xTestAccount1…',
  '0xTestAccount2…',
  '0xTestAccount3…',
]);
dev-setup.mjs · ethers v6
import { ethers } from 'ethers';

const RPC = {
  sepolia: 'https://rpc.sepolia.org',
  polygon: 'https://rpc-amoy.polygon.technology',
};

// Generate a fresh wallet and fund it in one call
async function createAndFund(network = 'sepolia') {
  const wallet = ethers.Wallet.createRandom();
  console.log('🔑 New address:', wallet.address);

  const res = await fetch(
    'https://sepolia-faucet-service.vercel.app/api/faucet',
    {
      method:  'POST',
      headers: { 'Content-Type': 'application/json' },
      body:    JSON.stringify({ address: wallet.address, network })
    }
  );
  const data = await res.json();
  if (!data.success) throw new Error(data.message);

  console.log(`💰 Funded: ${data.amount} ${data.currency}`);
  console.log(`🔗 TX:     ${data.explorerUrl}`);

  const provider = new ethers.JsonRpcProvider(RPC[network]);
  return wallet.connect(provider);
}

// Ready-to-use funded wallet for tests
const dev = await createAndFund('polygon');
console.log('✅ Wallet ready:', dev.address);

🚀 Quick Start

Get test ETH with a simple API call. No authentication required for normal mode.

cURL 
curl -X POST https://sepolia-faucet-service.vercel.app/api/faucet \
  -H "Content-Type: application/json" \
  -d '{"address": "0x742d35Cc6634C0532925a3b844Bc9e7595f0bEb"}'
JavaScript 
const response = await fetch('https://sepolia-faucet-service.vercel.app/api/faucet', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ address: '0x742d35Cc6634C0532925a3b844Bc9e7595f0bEb' })
});

const data = await response.json();
console.log(data.transactionHash);

🔌 Integration Examples

React Hook

JavaScript 
import { useState } from 'react';

export function useFaucet() {
  const [loading, setLoading] = useState(false);

  const requestETH = async (address) => {
    setLoading(true);
    try {
      const response = await fetch('https://sepolia-faucet-service.vercel.app/api/faucet', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ address })
      });
      return await response.json();
    } finally {
      setLoading(false);
    }
  };

  return { requestETH, loading };
}

Python

Python 
import requests

def request_test_eth(address):
    response = requests.post(
        'https://sepolia-faucet-service.vercel.app/api/faucet',
        json={'address': address}
    )
    return response.json()

result = request_test_eth('0x742d35Cc6634C0532925a3b844Bc9e7595f0bEb')
print(result['transactionHash'])
